string getHiddenPassword(bool showAsterisk = false) {
    string password;
    char ch;

    while (true) {
        ch = _getch();   // read single char (not shown on screen)

        if (ch == 13) {  // ENTER key pressed
            cout << endl;
            break;
        } 
        else if (ch == 8) {  // BACKSPACE
            if (!password.empty()) {
                password.pop_back();
                if (showAsterisk) cout << "\b \b";
            }
        } 
        else {
            password.push_back(ch);
            if (showAsterisk) cout << '*';  // show '*' if enabled
        }
    }
    return password;
}
string generateId(const string &filename, const string &prefix) {
    ifstream fin(filename);
    string line, lastId = "";
    
    while (getline(fin, line)) {
        stringstream ss(line);
        string id;
        getline(ss, id, ',');  // First column is ID
        lastId = id;
    }
    fin.close();

    int num = 0;
    if (!lastId.empty()) {
        // remove prefix T or S
        num = stoi(lastId.substr(1));
    }
    num++; // next ID
    stringstream ss;
    ss << prefix << setw(3) << setfill('0') << num;
    return ss.str();
}

// Function to validate password strength
bool isValidPassword(const string &password) {
    if (password.length() < 8) return false;

    bool hasUpper = false, hasLower = false, hasDigit = false, hasSpecial = false;

    for (size_t i = 0; i < password.length(); i++) {
        char c = password[i];
        if (isupper(c)) hasUpper = true;
        else if (islower(c)) hasLower = true;
        else if (isdigit(c)) hasDigit = true;
        else hasSpecial = true;
    }

    return hasUpper && hasLower && hasDigit && hasSpecial;
}
bool isValidEmail(const string &email) {
    int atPos = email.find('@');
    int dotPos = email.find('.', atPos);

    return (atPos != string::npos && dotPos != string::npos && atPos < dotPos);
}
bool isValidPhone(const string &phone) {
    if (phone.length() != 10) return false;
    for (int i = 0; i < phone.length(); i++) {
        if (!isdigit(phone[i])) return false;
    }
    return true;
}
bool isValidDOB(const string &dob) {
    if (dob.length() != 10) return false; // 10 chars: DD/MM/YYYY
    if (dob[2] != '/' || dob[5] != '/') return false;

    string dd = dob.substr(0,2);
    string mm = dob.substr(3,2);
    string yyyy = dob.substr(6,4);

    int day = stoi(dd);
    int month = stoi(mm);
    int year = stoi(yyyy);

    if (month < 1 || month > 12) return false;
    if (day < 1 || day > 31) return false;
    if (year < 1900 || year > 2025) return false;

    return true;
}
bool isValidGender(const string &gender) {
    return (gender == "Male" || gender == "Female" || gender == "Other");
}

bool isValidDeptId(const string &deptId, const vector<pair<string,string>> &deptList) {
    for (auto &d : deptList) {

bool existsInFile(const string &filename, const string &id) {
    ifstream fin(filename);
    string line;
    while (getline(fin, line)) {
        stringstream ss(line);
        string fileId;
        getline(ss, fileId, ',');
        if (fileId == id) return true;
    }
    return false;
}

// Load departments
vector<pair<string,string>> loadDepartments() {
    vector<pair<string,string>> list;
    ifstream fin("departments.csv");
    string line;
    while (getline(fin, line)) {
        if (line.empty()) continue;
        stringstream ss(line);
        string id, name, head;
        getline(ss, id, ',');
        getline(ss, name, ',');
        getline(ss, head, '\n');
        if (!id.empty()) list.emplace_back(id, name);
    }
    fin.close();
    return list;
}

// Load all courses
vector<Course> loadCourses() {
    vector<Course> list;
    ifstream fin("courses.csv");
    string line;
    while (getline(fin, line)) {
        if (line.empty()) continue;
        stringstream ss(line);
        Course c;
        getline(ss, c.id, ',');
        getline(ss, c.name, ',');
        getline(ss, c.deptId, '\n');
        if (!c.id.empty()) list.push_back(c);
    }
    fin.close();
    return list;
}
string trim(const string &s) {
    size_t start = s.find_first_not_of(" \t\r\n");
    size_t end = s.find_last_not_of(" \t\r\n");
    return (start == string::npos) ? "" : s.substr(start, end - start + 1);
}


vector<pair<string,string>> loadCoursesForDept(const string& deptId) {
    vector<pair<string,string>> courseList;
    ifstream fin("courses.csv");
    string line;
    while (getline(fin, line)) {
        stringstream ss(line);
        string courseId, courseName, courseDept;
        getline(ss, courseId, ',');
        getline(ss, courseName, ',');
        getline(ss, courseDept, ',');

        courseDept = trim(courseDept);  // <--- trim spaces

        if(courseDept == deptId) {
            courseList.push_back({trim(courseId), trim(courseName)});
        }
    }
    fin.close();
    return courseList;
}

// Get department name by ID
string getDepartmentName(const string &deptId) {
    auto depts = loadDepartments();
    for (auto &d : depts) if (d.first == deptId) return d.second;

}

string selectCourseForDept(const string& deptId) {

        cout << c.first << "\t" << c.second << "\n";
    }

    string choice;

    do {
        cout << "Enter Course ID: ";
        cin >> choice;
        for (auto &c : courses) {
            if (c.first == choice) {
                valid = true;
                break;
            }
        }
        if (!valid) cout << "[!] Invalid Course ID. Try again.\n";
    } while (!valid);

    return choice;
}

void printBanner(const std::string &title) {
    int width = 60;
    int leftMargin = (consoleWidth - width) / 2;
    std::string margin(leftMargin, ' ');

    std::cout << margin << std::string(width, '*') << "\n";
    std::cout << margin << "*" << std::string(width - 2, ' ') << "*\n";

    int innerMargin = 2;
    int padding = (width - 2 - title.size() - innerMargin * 2) / 2;
    std::cout << margin << "*"
              << std::string(padding, ' ') << std::string(innerMargin, ' ')
              << title
              << std::string(innerMargin, ' ')
              << std::string(width - 2 - title.size() - innerMargin * 2 - padding, ' ')
              << "*\n";

    std::cout << margin << "*" << std::string(width - 2, ' ') << "*\n";
    std::cout << margin << std::string(width, '*') << "\n\n";
}

        string enrollNo, rollNo, name, email, phone, dob, gender, deptId, courseId, address, pass;
        getline(ss, enrollNo, ','); getline(ss, rollNo, ','); getline(ss, name, ',');
        getline(ss, email, ','); getline(ss, phone, ','); getline(ss, dob, ','); getline(ss, gender, ',');
        getline(ss, deptId, ','); getline(ss, courseId, ','); getline(ss, address, ','); getline(ss, pass, '\n');
        studentMap[enrollNo] = make_tuple(name, rollNo, deptId, courseId);
    }
    fstu.close();

    // Load departments
    ifstream fdept("departments.csv");
    map<string, string> deptMap;
    while (getline(fdept, line)) {
        stringstream ss(line);
        string id, name;
        getline(ss, id, ','); getline(ss, name, ',');
        deptMap[id] = name;
    }
    fdept.close();

    // Load courses
    ifstream fcourse("courses.csv");
    map<string, tuple<string,string,string>> courseMap; // courseId -> (name, deptId, fee)
    while (getline(fcourse, line)) {
        stringstream ss(line);
        string id, cname, deptId, fee;
        getline(ss, id, ','); getline(ss, cname, ','); getline(ss, deptId, ','); getline(ss, fee, '\n');
        courseMap[id] = make_tuple(cname, deptId, fee);
    }
    fcourse.close();

    // Display fees with full details
    ifstream fin("fees.csv");
    cout << left << setw(8) << "FeeID" 
         << setw(12) << "EnrollNo" 
         << setw(20) << "Name" 
         << setw(20) << "Department"
         << setw(25) << "Course"
         << setw(10) << "Total"
         << setw(10) << "Paid"
         << setw(15) << "Date"
         << setw(10) << "Status" << "\n";
    cout << string(130, '-') << "\n";

    while (getline(fin, line)) {
        string feeId, stuId, courseId, total, paid, date, status;
        stringstream ss(line);
        getline(ss, feeId, ','); getline(ss, stuId, ','); getline(ss, courseId, ',');
        getline(ss, total, ','); getline(ss, paid, ','); getline(ss, date, ','); getline(ss, status, '\n');

        string name = (studentMap.count(stuId) ? get<0>(studentMap[stuId]) : "N/A");
        string deptName = (studentMap.count(stuId) ? deptMap[get<2>(studentMap[stuId])] : "N/A");
        string courseName = (courseMap.count(courseId) ? get<0>(courseMap[courseId]) : "N/A");

        cout << left << setw(8) << feeId 
             << setw(12) << stuId 
             << setw(20) << name 
             << setw(20) << deptName
             << setw(25) << courseName 
             << setw(10) << total 
             << setw(10) << paid 
             << setw(15) << date 
             << setw(10) << status << "\n";
    }
    fin.close();
}


void adminUpdateFeeStatus() {
    string feeId;
    cout << "Enter Fee ID to update: ";
    cin >> feeId;

    ifstream fin("fees.csv");
    ofstream fout("temp.csv");
    string line;
    bool found = false;

    while (getline(fin, line)) {
        stringstream ss(line);
        string fId, stuId, courseId, total, paid, date, status;
        getline(ss, fId, ','); getline(ss, stuId, ','); getline(ss, courseId, ',');
        getline(ss, total, ','); getline(ss, paid, ','); getline(ss, date, ','); getline(ss, status, '\n');

        if (fId == feeId) {
            found = true;
            cout << "Current Status: " << status << ", Amount Paid: " << paid << "\n";
            string newPaid, newDate, newStatus;
            do {
                cout << "Enter amount paid now: ";
                cin >> newPaid;
                if (!isNumber(newPaid)) cout << "[!] Enter valid number!\n";
            } while (!isNumber(newPaid));
            cout << "Enter payment date (DD/MM/YYYY): ";
            cin >> newDate;

            int totalAmt = stoi(total);
            int paidAmt = stoi(paid) + stoi(newPaid);

            if (paidAmt >= totalAmt) newStatus = "Paid";
            else if (paidAmt > 0) newStatus = "Partial";
            else newStatus = "Pending";

            paid = to_string(paidAmt);
            date = newDate;
            status = newStatus;
        }

        fout << fId << "," << stuId << "," << courseId << "," << total << "," 
             << paid << "," << date << "," << status << "\n";
    }

    fin.close();
    fout.close();
    remove("fees.csv");
    rename("temp.csv", "fees.csv");

    if (found) cout << "[+] Fee record updated successfully!\n";
    else cout << "[!] Fee ID not found!\n";
}
